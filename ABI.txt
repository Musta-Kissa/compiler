/* STACK LAYOUT
 * var 1            <- base ptr-40  ^
 * reg 4            <- base ptr-32  |
 * reg 3            <- base ptr-24  |
 * reg 2            <- base ptr-16  |
 * reg 1            <- base ptr-8   |  called function
 * saved base ptr   <- base ptr     v      
 * return addr      <- base ptr+8   ^
 * arg 1            <- base ptr+16  |
 * arg 2            <- base ptr+24  | caller function 
 * arg 3            <- base ptr+32  v
 * ...
*/

/*
 *  Non-Volotile registers: R8..R15
 * 
 *  Volotile registers: The rest
*/

/*
 * ABI
 * non volatile registers used for storing calculations: R8..R15
 * the rest are
 * volatile registers used for geting results from operations such as idiv
 * and returning values from fcalls (rax)
*/
